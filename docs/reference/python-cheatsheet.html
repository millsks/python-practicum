Comprehensions and generator expressions

[x * x for x in range(5)]                    # list
[x for x in nums if x % 2 == 0]              # filter
["even" if x % 2 == 0 else "odd" for x in nums]  # map with a condition
{w: len(w) for w in words}                   # dict
{c.lower() for c in text if c.isalpha()}     # set
(x * x for x in range(10**9))                # generator: lazy, O(1) memory
sum(x * x for x in nums)                     # no extra brackets needed
[(r, c) for r in range(3) for c in range(3) if r != c]   # nested: outer loop first
[x for row in matrix for x in row]           # flatten one level

def fib(n: int) -> int:

Classes and dataclasses

class Account:
    interest = 0.02                      # class attribute (shared)

    def __init__(self, owner: str, balance: float = 0) -> None:
        self.owner = owner                # instance attribute
        self._balance = balance           # "private" by convention

    @property
    def balance(self) -> float:           # read like an attribute
        return self._balance

    def deposit(self, amount: float) -> None:
        if amount <= 0:
            raise ValueError("amount must be positive")
        self._balance += amount

    @classmethod
    def empty(cls, owner: str) -> "Account":   # alternative constructor
        return cls(owner)

    @staticmethod
    def fee(amount: float) -> float:
        return amount * 0.01

    def __repr__(self) -> str:
        return f"Account({self.owner!r}, {self._balance})"

class Savings(Account):
    def __init__(self, owner: str) -> None:
        super().__init__(owner)

Async

import asyncio

async def fetch(name: str, delay: float) -> str:
    await asyncio.sleep(delay)              # yield to the event loop
    return name

async def main() -> None:
    a, b = await asyncio.gather(fetch("a", 1), fetch("b", 1))   # ~1 s total

    async with asyncio.TaskGroup() as tg:   # structured concurrency
        t1 = tg.create_task(fetch("x", 0.5))
        t2 = tg.create_task(fetch("y", 0.2))
    print(t1.result(), t2.result())

    async with asyncio.timeout(2):          # cancel if too slow
        await fetch("slow", 1)

    sem = asyncio.Semaphore(10)             # limit concurrency
    async with sem: ...

asyncio.run(main())                         # in a script (top-level await in the browser)
